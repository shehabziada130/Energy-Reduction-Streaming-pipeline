Total Energy Consumption by Zone: Bar guage basic 

	SELECT SUM("z1_AC2(kW)") + SUM("z1_Plug(kW)") + SUM("z1_AC3(kW)") AS "Zone 1 Total Consumption"
	FROM "anomaly_data"
	WHERE time > now() - 30d
	GROUP BY time(1h)
	FILL(null)
