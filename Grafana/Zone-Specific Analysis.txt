Feature Contribution to Anomalies:Bar chart 
	SELECT MEAN("z1_AC2(kW)") AS "Zone 1 AC2",
	       MEAN("z2_AC1(kW)") AS "Zone 2 AC1", 
	       MEAN("z3_Light(kW)") AS "Zone 3 Light", 
	       MEAN("z4_AC1(kW)") AS "Zone 4 AC1", 
	       MEAN("z5_AC1(kW)") AS "Zone 5 AC1"
	FROM "anomaly_data"
	WHERE "prediction" = 1
	GROUP BY time(10h)


	SELECT MEAN("z2_AC1(kW)") AS "Zone 2 AC1",

	       MEAN("z2_S1(lux)") AS "Zone 2 S1"
	FROM "anomaly_data" 
	GROUP BY time(1h) 
	FILL(null)
