Lux Level vs. Light Energy Consumption:Guage

	SELECT MEAN("z1_Plug(kW)") AS "Zone 1 Plug Consumption",
	       MEAN("z1_S1(lux)") AS "Zone 1 Lux Level"
	FROM "anomaly_data"
	GROUP BY time(1h)
	FILL(null)



	SELECT MEAN("z4_Light(kW)") AS "Zone 4 Light Consumption",
	       MEAN("z4_S1(lux)") AS "Zone 4 Lux Level"
	FROM "anomaly_data"
	GROUP BY time(1h)
	FILL(null)
